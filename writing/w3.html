#!/bin/bash

# Derive PROJECT_ID, ZONE, and REGION dynamically from instance metadata
export PROJECT_ID=$(curl -s http://metadata.google.internal/computeMetadata/v1/project/project-id -H "Metadata-Flavor: Google")
export ZONE=$(curl -s http://metadata.google.internal/computeMetadata/v1/instance/zone -H "Metadata-Flavor: Google" | cut -d/ -f4)
export REGION=$(echo "$ZONE" | sed 's/-.$//') # Derive region from zone

## Install Golang
sudo apt update -y
sudo apt-get install wget -y git -y
sudo chmod 777 /usr/local/
sudo wget https://go.dev/dl/go1.23.0.linux-amd64.tar.gz
sudo tar -C /usr/local -xzf go1.23.0.linux-amd64.tar.gz
export PATH=$PATH:/usr/local/go/bin

# Install ops agent
curl -sSO https://dl.google.com/cloudagents/add-google-cloud-ops-agent-repo.sh
sudo bash add-google-cloud-ops-agent-repo.sh --also-install
sudo service google-cloud-ops-agent start

# Create go working directory and add go path
mkdir -p /work/go/cache # -p creates parent directories if they don't exist
export GOPATH=/work/go
export GOCACHE=/work/go/cache

# Install Video queue Go source code
mkdir -p /work/go/video
gsutil cp gs://spls/gsp338/video_queue/main.go /work/go/video/main.go

# Navigate to the Go application directory before getting modules and running
cd /work/go/video

# Get Cloud Monitoring (stackdriver) modules
go get go.opencensus.io
go get contrib.go.opencensus.io/exporter/stackdriver

# Configure env vars for the Video Queue processing application dynamically
export MY_PROJECT_ID="$PROJECT_ID"
export MY_GCE_INSTANCE_ID=$(curl -s http://metadata.google.internal/computeMetadata/v1/instance/id -H "Metadata-Flavor: Google")
export MY_GCE_INSTANCE_ZONE="$ZONE"

# Initialize and run the Go application
go mod init video # Initialize module in the current directory
go mod tidy       # Clean up go.mod file
go run main.go    # Run the main application file
